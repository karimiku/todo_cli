```go
type ListFlags struct {
    JSON        bool  // --json
    Detail      bool  // -d / --detail
    Compact     bool  // --compact（将来用）
    MaxLen      int   // --maxlen N（将来用）
}

func parseListFlags(args []string) (ListFlags, []string, error) {
    flags := ListFlags{MaxLen: 60}
    remaining := []string{}
    
    for i := 0; i < len(args); i++ {
        switch args[i] {
        case "--json":
            flags.JSON = true
        case "-d", "--detail":
            flags.Detail = true
        case "--compact":
            flags.Compact = true
        case "--maxlen":
            if i+1 < len(args) {
                n, err := strconv.Atoi(args[i+1])
                if err == nil && n > 0 {
                    flags.MaxLen = n
                    i++
                }
            }
        default:
            remaining = append(remaining, args[i])
        }
    }
    
    return flags, remaining, nil
}
```

```go
type AddFlags struct {
    Quiet  bool  // -q / --quiet
    Urgent bool  // --urgent
    Stdin  bool  // 標準入力から読み取り（自動判定）
}

func parseAddFlags(args []string) (AddFlags, []string, error) {
    flags := AddFlags{}
    remaining := []string{}
    
    for _, arg := range args {
        switch arg {
        case "-q", "--quiet":
            flags.Quiet = true
        case "--urgent":
            flags.Urgent = true
        default:
            remaining = append(remaining, arg)
        }
    }
    
    return flags, remaining, nil
}
```

#### 3.2.2 標準入力対応

```go
func (c *AddCommand) Run(ctx *cli.CommandContext, args []string) error {
    flags, textArgs, err := parseAddFlags(args)
    if err != nil {
        return err
    }
    
    var text string
    
    // 引数がない場合は標準入力から読み取る
    if len(textArgs) == 0 {
        // 標準入力がパイプかどうかチェック
        stat, _ := os.Stdin.Stat()
        if (stat.Mode() & os.ModeCharDevice) == 0 {
            // パイプから読み取り
            scanner := bufio.NewScanner(ctx.Stdin)
            var lines []string
            for scanner.Scan() {
                lines = append(lines, scanner.Text())
            }
            if err := scanner.Err(); err != nil {
                return fmt.Errorf("標準入力の読み取りに失敗: %w", err)
            }
            text = strings.Join(lines, "\n")
        } else {
            return fmt.Errorf("usage: %s add <text> または標準入力からテキストを渡してください", ctx.BinaryName)
        }
    } else {
        text = strings.Join(textArgs, " ")
    }
    
    text = strings.TrimSpace(text)
    if text == "" {
        return storage.ErrEmptyText
    }
    
    // タスク追加
    var task *storage.Task
    if flags.Urgent {
        // 先頭に追加（AddTask + Focus の組み合わせ）
        task, err = ctx.Store.AddTask(ctx, text)
        if err != nil {
            return fmt.Errorf("タスク追加に失敗しました: %w", err)
        }
        // 先頭に移動
        _, err = ctx.Store.FocusTask(ctx, 1) // 追加されたタスクは末尾なので、その ID で focus
        if err != nil {
            return fmt.Errorf("先頭への移動に失敗しました: %w", err)
        }
    } else {
        task, err = ctx.Store.AddTask(ctx, text)
        if err != nil {
            return fmt.Errorf("タスク追加に失敗しました: %w", err)
        }
    }
    
    // 出力（quiet モードでは出力しない）
    if !flags.Quiet {
        summary := render.ExtractSummary(task.Text, 60)
        fmt.Fprintf(ctx.Stdout, "追加: %s\n", summary)
    }
    
    return nil
}
```

```go
package render

import (
    "strings"
    "unicode/utf8"

// ExtractSummary はテキストから要約を抽出します。
// 最初の行を取得し、Markdown ヘッダー記号を除去し、maxLen で切り詰めます。
func ExtractSummary(text string, maxLen int) string {
    // 改行で分割して最初の行を取得
    lines := strings.Split(text, "\n")
    firstLine := strings.TrimSpace(lines[0])
    
    // Markdown ヘッダー記号を除去
    firstLine = strings.TrimPrefix(firstLine, "# ")
    firstLine = strings.TrimPrefix(firstLine, "## ")
    firstLine = strings.TrimPrefix(firstLine, "### ")
    firstLine = strings.TrimPrefix(firstLine, "#### ")
    firstLine = strings.TrimPrefix(firstLine, "##### ")
    firstLine = strings.TrimPrefix(firstLine, "###### ")
    firstLine = strings.TrimSpace(firstLine)
    
    // 最大文字数で切り詰め（UTF-8 セーフ）
    return TruncateString(firstLine, maxLen)
}

// TruncateString は文字列を指定された文字数で切り詰めます（rune ベース）
func TruncateString(s string, maxLen int) string {
    runes := []rune(s)
    if len(runes) <= maxLen {
        return s
    }
    return string(runes[:maxLen]) + "…"
}

// ExtractTitle は要約をさらに短くしたタイトルを抽出します
func ExtractTitle(text string, maxLen int) string {

    // 現状はシンプルに切り詰めるだけ
    
    return summary
}
```

```go
package render

import (
    "github.com/charmbracelet/glamour"
)

// RenderMarkdown は Markdown をターミナル用にレンダリングします
func RenderMarkdown(content string, width int) (string, error) {
    renderer, err := glamour.NewTermRenderer(
        glamour.WithAutoStyle(),
        glamour.WithWordWrap(width),
    )
    if err != nil {
        return "", err
    }
    
    return renderer.Render(content)
}

// RenderMarkdownPreview はプレビュー用に高さ制限付きでレンダリングします
func RenderMarkdownPreview(content string, width, maxLines int) (string, error) {
    rendered, err := RenderMarkdown(content, width)
    if err != nil {
        return "", err
    }
    
    // 最大行数で切り詰め
    lines := strings.Split(rendered, "\n")
    if len(lines) > maxLines {
        lines = lines[:maxLines]
        lines = append(lines, "                                                           ↓ 続きあり")
    }
    
    return strings.Join(lines, "\n"), nil
}
```

```go
// ui/models.go

package ui

import (
    "github.com/charmbracelet/bubbles/list"
    "github.com/charmbracelet/bubbles/viewport"
    tea "github.com/charmbracelet/bubbletea"
    "github.com/kamiriku/todo_cli/internal/storage"
    "github.com/kamiriku/todo_cli/internal/render"
)

type mode int

const (
    modeList mode = iota     // リスト選択モード
    modeDetail               // 詳細表示モード
)

type model struct {
    list     list.Model
    viewport viewport.Model
    tasks    []storage.Task
    mode     mode
    width    int
    height   int
}

func newModel(pending, completed []storage.Task) model {
    // リストアイテムを作成
    items := make([]list.Item, len(pending))
    for i, task := range pending {
        items[i] = taskItem{
            id:      i + 1,
            task:    task,
            summary: render.ExtractSummary(task.Text, 60),
        }
    }
    
    // リストモデルを作成
    delegate := list.NewDefaultDelegate()
    l := list.New(items, delegate, 0, 0)
    l.Title = "📋 Todo List"
    l.SetShowStatusBar(false)
    l.SetFilteringEnabled(false)
    
    // viewport を作成
    vp := viewport.New(80, 20)
    
    return model{
        list:     l,
        viewport: vp,
        tasks:    pending,
        mode:     modeList,
    }
}

type taskItem struct {
    id      int
    task    storage.Task
    summary string
}

func (t taskItem) FilterValue() string { return t.summary }
func (t taskItem) Title() string       { return t.summary }
func (t taskItem) Description() string { return "" }
```

#### 3.5.3 Update ロジック

```go
// ui/models.go

func (m model) Init() tea.Cmd {
    return nil
}

func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
    switch msg := msg.(type) {
    case tea.WindowSizeMsg:
        m.width = msg.Width
        m.height = msg.Height
        
        // リストは上部 40%
        listHeight := int(float64(msg.Height) * 0.4)
        m.list.SetSize(msg.Width, listHeight)
        
        // viewport は下部 60%
        previewHeight := msg.Height - listHeight - 4 // ヘッダー分
        m.viewport.Width = msg.Width - 2
        m.viewport.Height = previewHeight
        
        // 初期プレビューを更新
        if m.mode == modeList {
            m.updatePreview()
        }
        
        return m, nil
        
    case tea.KeyMsg:
        switch msg.String() {
        case "ctrl+c":
            return m, tea.Quit
            
        case "q":
            if m.mode == modeDetail {
                // 詳細表示からリスト選択に戻る
                m.mode = modeList
                m.updatePreview()
                return m, nil
            }
            // リスト選択モードでは終了
            return m, tea.Quit
            
        case "enter":
            if m.mode == modeList {
                // 詳細表示モードに切り替え
                m.mode = modeDetail
                m.updateDetailView()
                return m, nil
            }
            
        case "j", "down":
            if m.mode == modeList {
                // リスト内で移動
                m.list, _ = m.list.Update(msg)
                m.updatePreview()
                return m, nil
            } else {
                // 詳細表示内でスクロール
                m.viewport, _ = m.viewport.Update(msg)
                return m, nil
            }
            
        case "k", "up":
            if m.mode == modeList {
                m.list, _ = m.list.Update(msg)
                m.updatePreview()
                return m, nil
            } else {
                m.viewport, _ = m.viewport.Update(msg)
                return m, nil
            }
            
        case "g":
            if m.mode == modeDetail {
                // 先頭へ
                m.viewport.GotoTop()
                return m, nil
            }
            
        case "G":
            if m.mode == modeDetail {
                // 末尾へ
                m.viewport.GotoBottom()
                return m, nil
            }
        }
    }
    
    return m, nil
}

func (m *model) updatePreview() {
    idx := m.list.Index()
    if idx >= 0 && idx < len(m.tasks) {
        task := m.tasks[idx]
        rendered, _ := render.RenderMarkdownPreview(task.Text, m.viewport.Width, m.viewport.Height)
        m.viewport.SetContent(rendered)
    }
}

func (m *model) updateDetailView() {
    idx := m.list.Index()
    if idx >= 0 && idx < len(m.tasks) {
        task := m.tasks[idx]
        rendered, _ := render.RenderMarkdown(task.Text, m.width-4)
        m.viewport.SetContent(rendered)
        m.viewport.GotoTop()
    }
}
```

```go
import (
    "github.com/charmbracelet/lipgloss"

func (m model) View() string {
    if m.mode == modeDetail {
        return m.renderDetailView()
    }
    return m.renderListView()
}

func (m model) renderListView() string {
    headerStyle := lipgloss.NewStyle().
        Bold(true).
        Foreground(lipgloss.Color("#FAFAFA")).
        Background(lipgloss.Color("#7D56F4")).
        Padding(0, 1)
    
    header := headerStyle.Render("j/k: 移動 | Enter: 詳細表示 | q: 終了")
    
    divider := lipgloss.NewStyle().
        Foreground(lipgloss.Color("#666666")).
        Render(strings.Repeat("─", m.width))
    
    previewHeader := lipgloss.NewStyle().
        Bold(true).
        Foreground(lipgloss.Color("#FAFAFA")).
        Render("📄 Preview")
    
    return lipgloss.JoinVertical(
        lipgloss.Left,
        header,
        m.list.View(),
        divider,
        previewHeader,
        m.viewport.View(),
        divider,
    )
}

func (m model) renderDetailView() string {
    headerStyle := lipgloss.NewStyle().
        Bold(true).
        Foreground(lipgloss.Color("#FAFAFA")).
        Background(lipgloss.Color("#7D56F4")).
        Padding(0, 1)
    
    idx := m.list.Index()
    header := headerStyle.Render(
        fmt.Sprintf("📄 Task #%d Details | q: 戻る | j/k: スクロール | gg: 先頭 | G: 末尾", idx+1),
    )
    
    divider := lipgloss.NewStyle().
        Foreground(lipgloss.Color("#666666")).
        Render(strings.Repeat("─", m.width))
    
    return lipgloss.JoinVertical(
```
